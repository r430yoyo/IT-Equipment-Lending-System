            const deviceVal = document.getElementById('filterResDevice')?.value || 'ALL';

            const userQuery = (document.getElementById('filterResUserName')?.value || '').trim().toLowerCase();

            const deptVal = document.getElementById('filterResDept')?.value || 'ALL';

            const dateQuery = document.getElementById('filterResDate')?.value || '';

 

            const effectiveStatus = window.currentResFilter || 'ALL';

 

            return window.reservations.filter(res => {

                if (effectiveStatus !== 'ALL') {

                    if (effectiveStatus === 'approved') {

                        if (res.status !== 'approved' && res.status !== 'pending') return false;

                    } else if (effectiveStatus === 'active') {

                        if (res.status !== 'active') return false;

                    } else if (effectiveStatus === 'completed') {

                        if (res.status !== 'completed') return false;

                    } else if (effectiveStatus === 'cancelled') {

                        if (res.status !== 'cancelled' && res.status !== 'rejected') return false;

                    }

                }

 

                const matchedLaptop = window.laptops.find(l => l.id === res.laptopId);

                const category = matchedLaptop?.category || '筆電';

                const code = res.laptopName || (matchedLaptop ? (matchedLaptop.code || matchedLaptop.id) : res.laptopId) || '';

 

                if (catVal !== 'ALL' && category !== catVal) return false;

                if (deviceVal !== 'ALL' && res.laptopId !== deviceVal && code !== deviceVal) return false;

 

                if (userQuery) {

                    const uName = (res.userName || '').toLowerCase();

                    const borrowUser = (res.actualBorrowUserName || '').toLowerCase();

                    const returnUser = (res.actualReturnUserName || '').toLowerCase();

                    if (!uName.includes(userQuery) && !borrowUser.includes(userQuery) && !returnUser.includes(userQuery)) return false;

                }

 

                if (deptVal !== 'ALL' && res.userDept !== deptVal) return false;

 

                if (dateQuery) {

                    if (res.startDate && res.endDate) {

                        if (dateQuery < res.startDate || dateQuery > res.endDate) return false;

                    }

                }

 

                return true;

            });

        };

        window.showToast = function(message, type = 'info') {

            const container = document.getElementById('toastContainer');

            if (!container) return;

 

            const toast = document.createElement('div');

            let iconClass = 'fa-circle-info text-sky-400';

            let themeClass = 'bg-slate-900/95 text-white border-slate-700 shadow-slate-900/40 ring-1 ring-white/10';

 

            if (type === 'success') {

                iconClass = 'fa-circle-check text-emerald-400';

                themeClass = 'bg-slate-900/95 text-white border-emerald-500/50 shadow-emerald-950/40 ring-1 ring-emerald-500/20';

            } else if (type === 'error') {

                iconClass = 'fa-circle-exclamation text-rose-400';

                themeClass = 'bg-slate-900/95 text-white border-rose-500/50 shadow-rose-950/40 ring-1 ring-rose-500/20';

            } else if (type === 'warning') {

                iconClass = 'fa-triangle-exclamation text-amber-400';

                themeClass = 'bg-slate-900/95 text-white border-amber-500/50 shadow-amber-950/40 ring-1 ring-amber-500/20';

            }

 

            toast.className = `pointer-events-auto flex items-center justify-center gap-4 px-8 py-5 rounded-2xl border shadow-2xl backdrop-blur-md text-base sm:text-lg font-semibold ${themeClass} animate-fade-in transition-all duration-300 w-full text-center`;

            toast.innerHTML = `

                <i class="fa-solid ${iconClass} text-2xl shrink-0"></i>

                <span class="leading-relaxed tracking-wide">${escapeHtml(message)}</span>

            `;

 

            container.appendChild(toast);

 

            setTimeout(() => {

                toast.style.opacity = '0';

                toast.style.transform = 'scale(0.92)';

                setTimeout(() => toast.remove(), 300);

            }, 3200);

        };

 

        function escapeHtml(str) {

                .replace(/"/g, '&quot;')

                .replace(/'/g, '&#039;');

        }
